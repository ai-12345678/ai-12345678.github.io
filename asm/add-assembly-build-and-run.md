## 2. add.asm 原始源码与结构

### 2.1 完整源码

将下面代码保存为 UTF-8 编码的 `add.asm`。代码、注释和标签均按用户提供的原文保留。

```nasm
; ============================================================
; 文件名称：add.asm
; 功能：从终端读取两个整数，计算相加结果并输出
;
; 运行平台：
;   CPU : x86-64
;   OS  : Linux
;   ABI : System V AMD64
;   ASM : NASM（Intel 语法）
;
; 对应的 C 语言逻辑：
;
;   int main(void) {
;       int a, b;
;       printf("请输入两个整数: ");
;
;       if (scanf("%d %d", &a, &b) != 2) {
;           return 1;
;       }
;
;       printf("相加结果: %d\n", a + b);
;       return 0;
;   }
;
; 注意：汇编的 32 位加法按模 2^32 计算。
;       C 语言有符号整数溢出则属于未定义行为。
; ============================================================


; ------------------------------------------------------------
; 一、声明全局符号和外部函数
; ------------------------------------------------------------

global main

; global：
;   将 main 导出为全局符号，供链接器查找。
;
; main：
;   程序的 C 语言入口函数。
;   真正的进程入口通常是 C 运行时的 _start，
;   C 运行时初始化后再调用 main。


extern printf
extern scanf

; extern：
;   声明函数在当前文件外部定义。
;
; printf：
;   C 标准库输出函数。
;   int printf(const char *format, ...);
;
; scanf：
;   C 标准库输入函数。
;   int scanf(const char *format, ...);
;
; 这两个函数由链接器从 C 标准库中解析。


; ------------------------------------------------------------
; 二、已初始化数据段：.data
; ------------------------------------------------------------

section .data

    ; db = Define Byte，定义字节数据。
    ;
    ; NASM 字符串按字节存储。
    ; 使用 UTF-8 编码保存源文件时，中文按 UTF-8 字节存储。
    ;
    ; 字符串末尾的 0 是 C 字符串结束标记 '\0'。

    prompt db "请输入两个整数: ", 0

    ; prompt 是字符串起始地址的符号。
    ; 内存保存：
    ;   UTF-8 编码的提示文本 + 00


    fmt_in db "%d %d", 0

    ; scanf 的格式字符串：
    ;   第一个 %d：读取一个有符号 int 整数。
    ;   第二个 %d：再读取一个有符号 int 整数。
    ;
    ; 两个 %d 之间的空格代表可匹配任意数量的空白字符。
    ; 例如输入：10 20


    fmt_out db "相加结果: %d", 10, 0

    ; printf 的格式字符串：
    ;   %d：输出一个有符号 int 整数。
    ;   10：ASCII LF，换行符 '\n'。
    ;   0 ：C 字符串结束标记 '\0'。
    ;
    ; 因此 printf 输出结果后会换行。


; ------------------------------------------------------------
; 三、未显式初始化数据段：.bss
; ------------------------------------------------------------

section .bss

    a resd 1
    b resd 1

    ; resd = Reserve Doubleword。
    ; 一个 doubleword = 4 字节 = 32 位。
    ;
    ; a：
    ;   为第一个 int 整数预留 4 字节。
    ;
    ; b：
    ;   为第二个 int 整数预留 4 字节。
    ;
    ; resd 1 表示预留一个 32 位整数的位置。
    ;
    ; .bss 在程序加载时会被初始化为零。
    ;
    ; 类似 C 语言的文件作用域变量：
    ;   int a;
    ;   int b;


; ------------------------------------------------------------
; 四、代码段：.text
; ------------------------------------------------------------

section .text

main:

    ; --------------------------------------------------------
    ; 1. 建立当前函数的栈帧
    ; --------------------------------------------------------

    push rbp

    ; push rbp：
    ;   1. RSP = RSP - 8
    ;   2. 将旧的 RBP 值写入 [RSP]
    ;
    ; RSP：栈指针寄存器，指向当前栈顶。
    ; RBP：这里用于保存当前函数的栈帧基址。
    ;
    ; x86-64 的栈通常向低地址方向增长。


    mov rbp, rsp

    ; mov 目标, 源：
    ;   将源操作数复制到目标操作数。
    ;
    ; RBP = RSP
    ;
    ; 此时建立了 main 函数的栈帧。
    ;
    ; System V ABI 要求调用其他函数之前，
    ; RSP 按 16 字节对齐。
    ;
    ; main 刚进入时，通常 RSP % 16 = 8。
    ; push rbp 后，RSP % 16 = 0。
    ;
    ; 因此接下来可以直接调用 printf / scanf。


    ; --------------------------------------------------------
    ; 2. 输出提示：printf("请输入两个整数: ");
    ; --------------------------------------------------------

    lea rdi, [rel prompt]

    ; lea = Load Effective Address。
    ;
    ; 此处将 prompt 的地址放入 RDI：
    ;
    ;   RDI = &prompt[0]
    ;
    ; rel 表示使用 RIP 相对寻址。
    ;
    ; System V ABI：
    ;   RDI 保存第一个整数或指针参数。
    ;
    ; printf 第一个参数是格式字符串指针。


    xor eax, eax

    ; xor：
    ;   对两个操作数执行按位异或。
    ;
    ; eax XOR eax = 0
    ; 因此 EAX = 0。
    ;
    ; printf 是可变参数函数。
    ; 在 System V AMD64 ABI 中，AL 用于提供
    ; 向量寄存器参数数量的上界。
    ;
    ; 当前没有通过向量寄存器传递浮点参数，
    ; 因此设置 AL = 0。
    ;
    ; xor eax, eax 同时会清零整个 RAX。


    call printf

    ; call：
    ;   1. 将下一条指令的返回地址压入栈。
    ;   2. 跳转到 printf 执行。
    ;
    ; printf 输出提示字符串。
    ;
    ; printf 返回后：
    ;   EAX 保存 printf 的返回值，
    ;   正常情况下是输出的字节数。
    ;
    ; 后续代码不使用该返回值。


    ; --------------------------------------------------------
    ; 3. 输入两个整数：scanf("%d %d", &a, &b);
    ; --------------------------------------------------------

    lea rdi, [rel fmt_in]

    ; RDI = fmt_in 字符串的起始地址。
    ; scanf 的第一个参数。


    lea rsi, [rel a]

    ; RSI = 变量 a 的内存地址。
    ;
    ; 注意：
    ;   传入的是 &a，而不是 a 的数值。
    ;
    ; scanf 需要通过地址修改内存中的变量。


    lea rdx, [rel b]

    ; RDX = 变量 b 的内存地址。
    ;
    ; System V ABI 前三个整数/指针参数：
    ;   第一个：RDI
    ;   第二个：RSI
    ;   第三个：RDX


    xor eax, eax

    ; scanf 同样属于可变参数函数。
    ; 将 AL 设置为 0，满足调用约定。


    call scanf

    ; scanf 等待并解析终端输入。
    ;
    ; 例如输入：
    ;   10 20
    ;
    ; 执行完成后：
    ;   内存 [a] = 10
    ;   内存 [b] = 20
    ;
    ; EAX 是 scanf 的返回值：
    ;   2  = 成功读取并赋值两个整数
    ;   1  = 只成功读取一个整数
    ;   0  = 没有成功匹配整数
    ;  -1  = 在首次成功转换前发生输入失败（EOF）


    ; --------------------------------------------------------
    ; 4. 检查 scanf 是否成功读取两个整数
    ; --------------------------------------------------------

    cmp eax, 2

    ; cmp = Compare。
    ;
    ; 在逻辑上执行：
    ;   EAX - 2
    ;
    ; 不保存减法结果，只更新 RFLAGS 标志位。
    ;
    ; 如果 EAX == 2：
    ;   ZF = 1（Zero Flag）
    ;
    ; 如果 EAX != 2：
    ;   ZF = 0


    jne error

    ; jne = Jump if Not Equal。
    ;
    ; 当 ZF == 0 时跳转到 error。
    ;
    ; 当 ZF == 1 时继续执行下一条指令。
    ;
    ; 等价于：
    ;   if (scanf_result != 2) goto error;


    ; --------------------------------------------------------
    ; 5. 将内存中的整数读取到 CPU 寄存器
    ; --------------------------------------------------------

    mov eax, [rel a]

    ; mov eax, [rel a]：
    ;
    ; 读取 a 地址处的 4 字节整数，
    ; 将该数值加载到 EAX。
    ;
    ; 例如：
    ;   内存 a = 10
    ;
    ; 执行后：
    ;   EAX = 10
    ;
    ; 注意区分：
    ;   lea rdi, [rel a]  ; 取得地址
    ;   mov eax, [rel a]  ; 读取地址中的数值


    ; --------------------------------------------------------
    ; 6. 执行加法：a + b
    ; --------------------------------------------------------

    add eax, [rel b]

    ; add 目标, 源：
    ;
    ;   目标 = 目标 + 源
    ;
    ; 此处：
    ;   EAX = EAX + 内存[b]
    ;
    ; 例如：
    ;   EAX = 10
    ;   内存 b = 20
    ;
    ; 执行后：
    ;   EAX = 30
    ;
    ; add 还会更新 RFLAGS 中的相关标志位。
    ;
    ; 此处使用 EAX，因此进行 32 位加法。
    ; 超出 32 位范围时仅保留结果的低 32 位。
    ;
    ; 此时计算结果保存在 EAX，不是内存 a 中。


    ; --------------------------------------------------------
    ; 7. 准备 printf 的输出参数
    ; --------------------------------------------------------

    lea rdi, [rel fmt_out]

    ; 第一个参数：
    ;   RDI = 格式字符串的地址。
    ;
    ; 格式字符串：
    ;   "相加结果: %d\n"


    mov esi, eax

    ; 第二个参数：
    ;   ESI = EAX
    ;
    ; 因为 printf 的第二个参数是 int，
    ; 根据 System V ABI，使用 ESI 传递。
    ;
    ; 例如：
    ;   EAX = 30
    ;   执行后 ESI = 30
    ;
    ; 这样加法结果就被传给 printf。


    xor eax, eax

    ; 将 AL 设置为 0，满足可变参数调用约定。
    ;
    ; 此时会覆盖 EAX 中原来的加法结果，
    ; 但结果已经复制到了 ESI，因此不会丢失。


    call printf

    ; 调用 printf。
    ;
    ; 最终输出：
    ;   相加结果: 30
    ;
    ; 返回后 EAX 保存 printf 的返回值。
    ; 这里不需要使用该值。


    ; --------------------------------------------------------
    ; 8. 正常退出：return 0;
    ; --------------------------------------------------------

    mov eax, 0

    ; main 返回值为 0：
    ;   表示程序成功结束。
    ;
    ; EAX 是 int 函数返回值寄存器。


    jmp end

    ; 无条件跳转到 end。
    ; 跳过下面的错误处理分支。


    ; --------------------------------------------------------
    ; 9. 错误分支：return 1;
    ; --------------------------------------------------------

error:

    mov eax, 1

    ; scanf 没有成功读取两个整数。
    ;
    ; 设置 main 返回值为 1。
    ; 表示本程序以错误状态结束。
    ;
    ; 这里没有打印错误信息。


    ; --------------------------------------------------------
    ; 10. 恢复栈帧并返回
    ; --------------------------------------------------------

end:

    pop rbp

    ; pop rbp：
    ;   1. 从 [RSP] 读取原来的 RBP。
    ;   2. RSP = RSP + 8。
    ;
    ; 恢复调用者的 RBP。


    ret

    ; ret：
    ;   从 [RSP] 取出返回地址。
    ;   RSP = RSP + 8。
    ;   将控制流转回调用 main 的 C 运行时代码。
    ;
    ; main 的返回值保存在 EAX。
```

### 2.2 阅读源码时，先区分四类内容

| 类别 | 本例中的内容 | 谁处理、起什么作用 |
|---|---|---|
| 注释 | 以 `;` 开头或位于 `;` 后的文字 | 供人阅读，汇编器忽略，不生成机器指令 |
| 汇编器指令与伪指令 | `global`、`extern`、`section`、`db`、`resd` | NASM 用来组织符号、代码和数据，不是 CPU 执行的指令 |
| 符号与标签 | `main`、`prompt`、`a`、`b`、`error`、`end` | 为函数入口、数据位置或跳转位置命名 |
| CPU 指令 | `push`、`mov`、`lea`、`xor`、`call`、`cmp`、`jne`、`add`、`jmp`、`pop`、`ret` | 汇编为机器码，由 CPU 执行 |

例如，`section .text` 告诉 NASM“后面的指令放进代码节”；`main:` 标记位置；`push rbp` 才是一条 CPU 指令。标签不是变量的值，CPU 执行时使用的是最终编码的地址、偏移量与操作数。

### 2.3 源文件的组织

| 源码部分 | 定义或声明的内容 | 与后续执行的关系 |
|---|---|---|
| `global main` | 导出 `main` 符号 | C 运行时能引用本文件定义的 `main` |
| `extern printf`、`extern scanf` | 声明外部函数 | 调用目标由链接和动态加载过程处理 |
| `.data` | 三个以 `0` 结尾的字符串 | 为 `printf`、`scanf` 提供文本或格式字符串 |
| `.bss` | `a`、`b`，各预留 4 字节 | `scanf` 将输入值写到这些地址 |
| `.text` | `main`、`error`、`end` 下的指令 | 建立栈帧、调用函数、计算、跳转和返回 |

`.data` 中的定义顺序，不是“先执行提示文本，再执行格式字符串”。字符串是被代码引用的数据；真正的执行顺序由 `.text` 中的指令和控制流决定。

## 3. 汇编与链接

在 `add.asm` 所在目录执行：

```bash
# 汇编：把源码转换为 64 位 ELF 可重定位目标文件。
nasm -f elf64 add.asm -o add.o

# 链接：由 GCC 驱动链接器，加入 C 启动文件和标准库依赖。
# -no-pie 适配原源码的普通外部函数调用写法。
# -Wl,-z,noexecstack 将“不需要可执行栈”的选项传给链接器。
gcc -no-pie -Wl,-z,noexecstack add.o -o add
```

| 阶段 | 输入 → 输出 | 必要理解 |
|---|---|---|
| 汇编 | `add.asm` → `add.o` | NASM 生成机器码、数据、符号表和重定位记录；`add.o` 不能直接运行 |
| 链接 | `add.o` 与启动文件、库 → `add` | GCC 驱动链接器，生成可执行文件；这里 GCC 不重新编译 NASM 源码 |

原源码没有 `.note.GNU-stack` 声明，因此在链接命令中显式指定 `noexecstack`，无需修改源码。`printf`、`scanf` 通常采用动态链接，其完整实现不必复制进 `add`。

## 5. 源码的执行流程与关键指令

### 5.1 从进程入口到 main

本例由 C 运行时启动。对通常的动态链接构建，Linux 与动态链接器先完成加载工作，随后进入可执行文件的 `_start`，C 运行时组织初始化并调用 `main`。

CPU 执行的是 NASM 生成、链接并加载到内存中的机器码。它不会在运行时读取源码注释，也不会直接解释 `printf` 这个名字；函数调用最终使用链接或动态绑定后的目标地址。

`main` 最后的 `ret` 返回 C 运行时，运行时处理正常退出并将退出状态交给操作系统。`ret` 本身不是退出进程的系统调用。

### 5.2 按源码顺序理解 main

| 步骤 | 对应指令 | 执行后发生什么 |
|---|---|---|
| 1. 建立栈帧 | `push rbp`、`mov rbp, rsp` | 保存旧 `RBP`，设置当前栈帧基址，使本例的后续调用前栈保持对齐 |
| 2. 打印提示 | `lea rdi, [rel prompt]`、`xor eax, eax`、`call printf` | 将提示地址作为第一个参数，调用输出函数 |
| 3. 读取输入 | 三条 `lea`、`xor eax, eax`、`call scanf` | 传入格式字符串、`&a`、`&b`；函数把输入写入内存 |
| 4. 检查输入 | `cmp eax, 2`、`jne error` | 若赋值数量不是两个，跳转到错误分支 |
| 5. 读取第一个整数 | `mov eax, [rel a]` | 读取 `a` 的 4 字节数值到 `EAX` |
| 6. 相加 | `add eax, [rel b]` | 加上 `b`，结果保存在 `EAX`；内存中的 `a`、`b` 不变 |
| 7. 输出结果 | `lea rdi, [rel fmt_out]`、`mov esi, eax`、`xor eax, eax`、`call printf` | 先将结果复制到 `ESI`，再设置可变参数调用信息并输出 |
| 8. 正常分支 | `mov eax, 0`、`jmp end` | 设置返回值 `0`，跳过错误分支 |
| 9. 错误分支 | `error:` 下的 `mov eax, 1` | 设置返回值 `1`，随后顺序执行到 `end` |
| 10. 返回 | `pop rbp`、`ret` | 恢复旧 `RBP`，根据返回地址回到调用者 |

`error:` 和 `end:` 是同一个 `main` 内部的跳转标签，并不是通过 `call` 调用的两个独立函数。正常路径与错误路径都会执行统一的收尾代码。

### 5.3 lea、mov 与内存寻址

| 原源码中的写法 | 实际含义 | 是否读取该地址处的数据 |
|---|---|---|
| `lea rsi, [rel a]` | 把 `a` 的地址放入 `RSI` | 否 |
| `mov eax, [rel a]` | 从 `a` 的地址读取 4 字节，放入 `EAX` | 是 |
| `add eax, [rel b]` | 读取 `b` 的 4 字节值，加到 `EAX` | 是 |
| `mov esi, eax` | 把寄存器中的结果复制到另一个寄存器 | 不访问内存 |

**取地址与读数值是两种不同操作。** `scanf` 需要变量地址，因为它要修改变量；加法需要变量的数值，所以先读取内存。

`[rel a]` 中的 `rel` 指定 RIP 相对寻址，实际地址通过该指令之后的 RIP 加上编码的位移得到。它是操作数的寻址方式，不是一条独立的 CPU 指令。`lea` 只计算地址，即使操作数写在方括号里，也不会读取该地址中的数值。

### 5.4 cmp、标志位与跳转

`cmp eax, 2` 按减法规则更新标志位，但不把减法结果写回 `EAX`。相等时 `ZF = 1`，不相等时 `ZF = 0`。紧随其后的 `jne error` 检查 `ZF`，决定是否跳转。

本例在 `cmp` 与 `jne` 之间没有插入其他修改标志位的指令，因此跳转依据就是对 `scanf` 返回值的比较。`jmp end` 是无条件跳转，不检查 `ZF`。

`add` 也会更新标志位，包括有符号溢出标志 `OF`，但本程序没有检查它。32 位结果溢出时保留低 32 位，仍可能正常返回 `0`；例如两个合法输入 `2147483647`、`1` 相加会输出 `-2147483648`。输入本身应位于 32 位有符号整数范围内。

### 5.5 三次库函数调用的参数

| 调用 | `RDI`：第一个参数 | `RSI` / `ESI`：第二个参数 | `RDX`：第三个参数 | 调用前的 `AL` |
|---|---|---|---|---|
| 提示输出 `printf` | `prompt` 的地址 | 无 | 无 | `0` |
| 读取输入 `scanf` | `fmt_in` 的地址 | `a` 的地址，使用 `RSI` | `b` 的地址 | `0` |
| 结果输出 `printf` | `fmt_out` 的地址 | 计算结果，使用 `ESI` 传递 `int` | 无 | `0` |

`printf`、`scanf` 都是可变参数函数。System V AMD64 ABI 用 `AL` 提供向量寄存器参数数量的上界；本例没有这类参数，因此每次调用前执行 `xor eax, eax`。

结果输出前的顺序很关键：先 `mov esi, eax` 保存结果，再 `xor eax, eax` 清零。如果颠倒这两条指令，传入的结果就变成 `0`。

这些库函数可以修改调用者保存寄存器，如 `RAX`、`RDI`、`RSI`、`RDX`。本源码每次调用前都会重新准备参数；`scanf` 返回后，`EAX` 保存的是赋值数量，而不是输入的整数。

## 6. 内存组织、寄存器与栈

### 6.1 代码、数据、栈与堆

| 区域 | 本程序中的内容 | 用途 |
|---|---|---|
| `.text` | `main` 下的指令，以及 `error`、`end` 标签处的指令 | 通常映射为可读、可执行的代码 |
| `.data` | `prompt`、`fmt_in`、`fmt_out` | 已初始化数据；字符串末尾 `0` 标记结束，`10` 表示换行 |
| `.bss` | `a`、`b`，各 4 字节 | 静态存储，初始为零，保存 `scanf` 写入的两个整数 |
| 栈 | 保存的旧 `RBP`、函数调用返回地址等 | 随调用和返回变化，通常向低地址增长 |
| 堆 | 本源码没有显式申请的堆对象 | 库函数或 C 运行时内部仍可能使用堆 |

源码中的 `.text`、`.data`、`.bss` 严格说是 ELF 的 **节（section）**；链接器将这些节组织到 **加载段（segment）** 中，运行时按段建立内存映射。`.bss` 通常记录空间大小，而不在文件中存放对应数量的零字节。

原源码注释中的 C 示例展示相同的输入、计算和输出流程，但其 `int a, b;` 是函数局部变量；汇编中的 `a`、`b` 则是 `.bss` 中的静态变量。两者存储组织不同，C 的有符号加法溢出语义也与本例的机器加法不同。

本函数建立了栈帧，却没有为 `a`、`b` 分配栈空间，也没有使用 `[rbp - ...]` 访问这两个变量。

### 6.2 寄存器不是独立变量名

| 寄存器 | 位宽 | 本例中的作用 |
|---|---|---|
| `RAX` / `EAX` / `AL` | 64 / 32 / 8 位 | 同一寄存器的不同视图；保存函数返回值、计算结果或可变参数调用信息 |
| `RDI` | 64 位 | 第一个参数，保存字符串地址 |
| `RSI` / `ESI` | 64 / 32 位 | 第二个参数，分别用于传递地址或 `int` 数值 |
| `RDX` | 64 位 | 第三个参数，传递 `b` 的地址 |
| `RSP` | 64 位 | 栈顶地址 |
| `RBP` | 64 位 | 本函数的栈帧基址，进入时保存，退出时恢复 |
| `RIP` | 64 位 | 指令指针，参与控制流和 RIP 相对寻址 |
| `RFLAGS` | 64 位标志寄存器 | 保存 `ZF`、`OF` 等运算与比较标志 |

写入 `EAX` 会清零 `RAX` 的高 32 位，所以 `xor eax, eax` 清零整个 `RAX`。写入 `ESI` 也会清零 `RSI` 的高 32 位；`printf` 的 `%d` 使用低 32 位作为有符号 `int`，因此本例传递负数也不必先把整个 `RSI` 符号扩展为 64 位。

同一个 `EAX` 在不同时刻有不同含义：`scanf` 返回后是赋值数量，读取 `a` 后是输入值，加法后是结果，最后是 `main` 的返回值。汇编不会为这些用途自动分配不同的变量。

### 6.3 栈对齐与返回地址

本例的调用没有通过栈传递额外参数，也没有分配局部栈空间，主要栈变化如下：

| 时刻 | 栈中的关键内容 | `RSP % 16` |
|---|---|---|
| 刚进入 `main` | 栈顶是返回 C 运行时的地址 | `8` |
| 执行 `push rbp` 后 | 栈顶是旧 `RBP`，上方是 `main` 返回地址 | `0` |
| 执行任意一次 `call` 后，进入库函数 | 新栈顶是返回 `main` 的地址 | `8` |
| 库函数返回到 `main` 后 | 恢复为调用前的栈顶 | `0` |
| 最后执行 `pop rbp` 后 | 恢复旧 `RBP`，栈顶是 `main` 返回地址 | `8` |
| 执行 `ret` 后 | 消耗 `main` 返回地址，控制流回到调用者 | 由调用者继续管理 |

System V AMD64 ABI 要求本例中每次 `call` **执行之前**栈顶按 16 字节对齐。`call` 自身再压入 8 字节返回地址，所以被调用函数进入时是 `RSP % 16 = 8`，这两点并不冲突。

`mov rbp, rsp` 只复制地址，不额外分配栈空间。原代码能直接使用 `pop rbp` 收尾，是因为自己的 `push`、库函数调用和返回整体保持平衡，结束时 `RSP` 仍指向保存的旧 `RBP`。
